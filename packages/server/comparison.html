レスポンス送信

async/http
async fn send_json(
  conn : @http.ServerConnection,
  code : Int, body : String,
) -> Unit {
  let headers : @http.Headers = Map([
    (@http.CaseInsensitiveString(
      "Content-Type"),
     "application/json"),
    (@http.CaseInsensitiveString(
      "Content-Length"),
     body.length().to_string()),
  ])
  conn.send_response(
    code, "OK",
    extra_headers=headers)
  conn.write_string(body)
  conn.end_response()
}

エラーハンドリング

MoonBit / mars

MoonBit の catch でエラーをステータスコードにマップ。

app.get("/users/:id", async fn(ctx) {
  let id = ctx.param("id")
  let user = find_user(id) catch {
    NotFound => {
      ctx.status(404)
      ctx.json("{\"error\":\"not found\"}")
      return
    }
    e => {
      ctx.status(500)
      ctx.json("{\"error\":\"internal\"}")
      return
    }
  }
  ctx.json(user.to_json())
})
axum

IntoResponse を実装した独自エラー型を返すだけ。

#[derive(thiserror::Error, Debug)]
enum AppError {
    #[error("not found")]
    NotFound,
    #[error("internal error")]
    Internal(#[from] anyhow::Error),
}
impl IntoResponse for AppError {
    fn into_response(self) -> Response {
        let status = match self {
            Self::NotFound => StatusCode::NOT_FOUND,
            Self::Internal(_) =>
              StatusCode::INTERNAL_SERVER_ERROR,
        };
        (status, self.to_string()).into_response()
    }
}
